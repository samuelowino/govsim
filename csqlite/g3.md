## Opening and Closing a Database Connection

### Basic Connection Management

The foundation of any SQLite application is opening and closing database connections. Let's start with the basics:

```swift
import CSQLite

class DatabaseConnection {
    private var db: OpaquePointer?
    
    func open(path: String) throws {
        // sqlite3_open: Opens a connection to the database
        // - Creates the database file if it doesn't exist
        // - Returns SQLITE_OK on success
        let result = sqlite3_open(path, &db)
        
        guard result == SQLITE_OK else {
            // Get detailed error message
            let errorMessage = String(cString: sqlite3_errmsg(db))
            throw DatabaseError.openFailed(errorMessage)
        }
        
        print("✅ Database opened at: \(path)")
    }
    
    func close() throws {
        guard let db = db else { 
            throw DatabaseError.notConnected 
        }
        
        // sqlite3_close: Closes the database connection
        // - Finalizes all prepared statements
        // - Returns SQLITE_OK on success
        let result = sqlite3_close(db)
        
        guard result == SQLITE_OK else {
            let errorMessage = String(cString: sqlite3_errmsg(db))
            throw DatabaseError.closeFailed(errorMessage)
        }
        
        self.db = nil
        print("✅ Database closed")
    }
    
    // Helper to check if connection is open
    var isOpen: Bool { db != nil }
}
```

```swift
        )
    }
```

## Executing SQL Statements

### Using sqlite3_exec for Simple Statements

`sqlite3_exec()` is the simplest way to execute SQL that doesn't return data:

```swift
class DatabaseExecutor {
    private var db: OpaquePointer?
    
    func execute(_ sql: String) throws {
        var errMsg: UnsafeMutablePointer<CChar>?
        defer { sqlite3_free(errMsg) }
        
        // sqlite3_exec: Execute one or more SQL statements
        let result = sqlite3_exec(db, sql, nil, nil, &errMsg)
        
        guard result == SQLITE_OK else {
            let error = errMsg.map { String(cString: $0) } ?? "Unknown error"
            throw DatabaseError.executeFailed(error)
        }
    }
}
```

## Working with Prepared Statements

### Basic Prepared Statement

Prepared statements are safer and more efficient for repeated operations:

```swift
class PreparedStatementExample {
    private var db: OpaquePointer?
    
    func insertUser(username: String, email: String, age: Int?) throws {
        let sql = "INSERT INTO users (username, email, age) VALUES (?, ?, ?)"
        var stmt: OpaquePointer?
        
        // 1. Prepare the statement
        guard sqlite3_prepare_v2(db, sql, -1, &stmt, nil) == SQLITE_OK else {
            let error = String(cString: sqlite3_errmsg(db))
            throw DatabaseError.prepareFailed(error)
        }
        defer { sqlite3_finalize(stmt) } // Always cleanup
        
        // 2. Bind parameters
        sqlite3_bind_text(stmt, 1, username, -1, SQLITE_TRANSIENT)
        sqlite3_bind_text(stmt, 2, email, -1, SQLITE_TRANSIENT)
        
        if let age = age {
            sqlite3_bind_int(stmt, 3, Int32(age))
        } else {
            sqlite3_bind_null(stmt, 3)
        }
        
        // 3. Execute
        guard sqlite3_step(stmt) == SQLITE_DONE else {
            let error = String(cString: sqlite3_errmsg(db))
            throw DatabaseError.executeFailed(error)
        }
        
        print("✅ User inserted with prepared statement")
    }
}
```

### Reusing Prepared Statements

```swift
class StatementCache {
    private var db: OpaquePointer?
    private var cache: [String: OpaquePointer] = [:]
    
    func getStatement(sql: String) throws -> OpaquePointer {
        // Return cached statement if available
        if let stmt = cache[sql] {
            sqlite3_reset(stmt) // Reset for reuse
            return stmt
        }
        
        // Create new statement
        var stmt: OpaquePointer?
        guard sqlite3_prepare_v2(db, sql, -1, &stmt, nil) == SQLITE_OK else {
            let error = String(cString: sqlite3_errmsg(db))
            throw DatabaseError.prepareFailed(error)
        }
        
        cache[sql] = stmt
        return stmt!
    }
    
    func clearCache() {
        for (_, stmt) in cache {
            sqlite3_finalize(stmt)
        }
        cache.removeAll()
    }
    
    deinit {
        clearCache()
    }
}
```

## Error Handling Patterns

### Comprehensive Error Handling

```swift
enum DatabaseError: LocalizedError {
    case openFailed(String)
    case closeFailed(String)
    case prepareFailed(String)
    case executeFailed(String)
    case stepFailed(String)
    case bindFailed(String)
    case constraintViolation(String)
    case notConnected
    case transactionFailed(String)
    
    var errorDescription: String? {
        switch self {
        case .openFailed(let msg): return "Failed to open database: \(msg)"
        case .closeFailed(let msg): return "Failed to close database: \(msg)"
        case .prepareFailed(let msg): return "Failed to prepare statement: \(msg)"
        case .executeFailed(let msg): return "Failed to execute: \(msg)"
        case .stepFailed(let msg): return "Failed to step: \(msg)"
        case .bindFailed(let msg): return "Failed to bind parameter: \(msg)"
        case .constraintViolation(let msg): return "Constraint violation: \(msg)"
        case .notConnected: return "Database not connected"
        case .transactionFailed(let msg): return "Transaction failed: \(msg)"
        }
    }
}

class RobustDatabase {
    private var db: OpaquePointer?
    
    func executeSafely(_ operation: () throws -> Void) {
        do {
            try operation()
        } catch let error as DatabaseError {
            print("⚠️ Database error: \(error.localizedDescription)")
            // Could log or report error
        } catch {
            print("⚠️ Unexpected error: \(error)")
        }
    }
    
    func safeQuery(sql: String) -> [String: Any]? {
        var stmt: OpaquePointer?
        defer { sqlite3_finalize(stmt) }
        
        guard sqlite3_prepare_v2(db, sql, -1, &stmt, nil) == SQLITE_OK else {
            return nil
        }
        
        if sqlite3_step(stmt) == SQLITE_ROW {
            var result: [String: Any] = [:]
            let columnCount = sqlite3_column_count(stmt)
            
            for i in 0..<columnCount {
                let name = String(cString: sqlite3_column_name(stmt, i))
                let type = sqlite3_column_type(stmt, i)
                
                switch type {
                case SQLITE_INTEGER:
                    result[name] = Int(sqlite3_column_int64(stmt, i))
                case SQLITE_FLOAT:
                    result[name] = Double(sqlite3_column_double(stmt, i))
                case SQLITE_TEXT:
                    result[name] = String(cString: sqlite3_column_text(stmt, i))
                case SQLITE_BLOB:
                    let blob = sqlite3_column_blob(stmt, i)
                    let size = sqlite3_column_bytes(stmt, i)
                    if let blob = blob, size > 0 {
                        result[name] = Data(bytes: blob, count: Int(size))
                    }
                case SQLITE_NULL:
                    result[name] = NSNull()
                default:
                    break
                }
            }
            return result
        }
        return nil
    }
}
```

### Retry Logic

```swift
class RetryableDatabase {
    private var db: OpaquePointer?
    private let maxRetries = 3
    
    func executeWithRetry(_ sql: String) throws {
        var lastError: Error?
        
        for attempt in 0..<maxRetries {
            do {
                try execute(sql)
                return // Success
            } catch let error as DatabaseError {
                lastError = error
                
                // Only retry on busy/locked errors
                if case .executeFailed(let msg) = error,
                   msg.contains("busy") || msg.contains("locked") {
                    print("⚠️ Database busy, retry \(attempt + 1)/\(maxRetries)")
                    Thread.sleep(forTimeInterval: 0.1 * Double(attempt + 1))
                    continue
                }
                throw error // Non-retryable error
            }
        }
        
        throw lastError ?? DatabaseError.executeFailed("Max retries exceeded")
    }
}
```

```swift
        print("✅ Statements prepared")

        var stmt: OpaquePointer?
        defer { sqlite3_finalize(stmt) }

            let error = String(cString: sqlite3_errmsg(db))

        let email = String(cString: sqlite3_column_text(stmt, 2))

            age = nil
        }

        let createdAt = Date(timeIntervalSince1970: TimeInterval(sqlite3_column_int64(stmt, 5)))

            id: id,
            username: username,
            email: email,
            age: age,
            isActive: isActive,
            createdAt: createdAt
        )

        let result = sqlite3_exec(db, sql, nil, nil, &errMsg)
        
        guard result == SQLITE_OK else {
            let error = errMsg.map { String(cString: $0) } ?? "Unknown error"
            throw DatabaseError.executeFailed(error)
        }
    }
}
```

```swift
class DatabaseConfigurator {
    private var db: OpaquePointer?
    
    func configureDatabase() throws {
        let pragmas: [(String, String)] = [
            ("journal_mode", "WAL"),           // Write-Ahead Logging for better concurrency
```
