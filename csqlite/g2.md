# Chapter 2: Installation and Setup

## Adding CSQLite to Your Project

### Option 1: Swift Package Manager (SPM)

The most common way to add CSQLite is through Swift Package Manager.

**In Package.swift:**
```swift
// swift-tools-version:6.0
import PackageDescription

let package = Package(
    name: "MyProject",
    dependencies: [
        .package(
            url: "https://github.com/sbooth/CSQLite.git",
            from: "1.0.0"
        )
    ],
    targets: [
        .target(
            name: "MyProject",
            dependencies: ["CSQLite"]
        )
    ]
)
```

**With specific version:**
```swift
// Exact version
.package(url: "https://github.com/sbooth/CSQLite.git", exact: "1.0.0")

// Up to next major (recommended)
.package(url: "https://github.com/sbooth/CSQLite.git", from: "1.0.0")

// Branch
.package(url: "https://github.com/sbooth/CSQLite.git", branch: "main")
```

### Option 2: Xcode Package Manager

1. Open your Xcode project
2. Go to **File → Add Package Dependencies...**
3. Enter the URL: `https://github.com/sbooth/CSQLite.git`
4. Click **Add Package**
5. Choose the target where you want to use CSQLite

### Option 3: Manual Installation

If you need to manually build CSQLite:

```bash
git clone https://github.com/sbooth/CSQLite.git
cd CSQLite
swift build
```

## Understanding Package Traits (Swift 6.1+)

Package traits allow you to customize how CSQLite is built. Let's explore each trait category in detail.

### Complete Trait Reference

#### 1. Double-Quoted String Handling (Choose ONE)

These traits control how SQLite handles double-quoted strings, which can be a source of SQL injection vulnerabilities.

```swift
// Default: SQLITE_DQS=0 (Strict, Recommended)
// Disallows "quoted" identifiers. Only single quotes for strings.
.package(
    url: "https://github.com/sbooth/CSQLite.git",
    from: "1.0.0",
    traits: ["DQS_0"]
)

// SQLITE_DQS=1 (Limited)
// Allows double-quoted strings in limited contexts
.package(..., traits: ["DQS_1"])

// SQLITE_DQS=2 (Permissive)
// Allows double-quoted strings as identifiers
.package(..., traits: ["DQS_2"])

// SQLITE_DQS=3 (Very Permissive)
// SQLite default, allows double-quoted strings as string literals
.package(..., traits: ["DQS_3"])
```

**Security Impact:**
```sql
-- With DQS_0 (Recommended)
SELECT * FROM "users";  -- ❌ Error: "users" is treated as string, not identifier
SELECT * FROM users;    -- ✅ Correct

-- With DQS_3 (SQLite default)
SELECT * FROM "users";  -- ✅ Works, but can mask SQL injection issues
```

#### 2. Threading Mode (Choose ONE)

```swift
// Default: SQLITE_THREADSAFE=0 (Single-threaded, Fastest)
// No thread-safety overhead. Must use from single thread.
.package(
    url: "https://github.com/sbooth/CSQLite.git",
    from: "1.0.0",
    traits: ["THREADSAFE_0"]
)

// SQLITE_THREADSAFE=1 (Serialized, Thread-safe)
// Full thread-safety with locking overhead
.package(..., traits: ["THREADSAFE_1"])

// SQLITE_THREADSAFE=2 (Multi-threaded, Limited)
// Can be used in multiple threads but without serialization
.package(..., traits: ["THREADSAFE_2"])
```

**Choosing the Right Mode:**

```swift
// 🚀 For single-threaded apps (e.g., command-line tools)
traits: ["THREADSAFE_0"]

// 🔒 For multi-threaded apps with shared database (e.g., web servers)
traits: ["THREADSAFE_1"]

// ⚖️ For multi-threaded apps with per-connection isolation
traits: ["THREADSAFE_2"]
```

#### 3. Performance and Behavior Options

```swift
// Disable memory status tracking (Default: Y)
// Slightly improves performance by removing memory counting
traits: ["DEFAULT_MEMSTATUS_0"]

// WAL synchronous mode = NORMAL (Default: Y)
// Good balance of safety and performance
traits: ["DEFAULT_WAL_SYNCHRONOUS_1"]

// LIKE operator doesn't match BLOBs (Default: Y)
// Prevents unexpected behavior
traits: ["LIKE_DOESNT_MATCH_BLOBS"]

// Remove expression depth limit (Default: Y)
// Allows deeply nested expressions
traits: ["MAX_EXPR_DEPTH_0"]

// Omit deprecated functions (Default: Y)
// Cleaner API surface
traits: ["OMIT_DEPRECATED"]

// Omit shared cache support (Default: Y)
// Reduces complexity
traits: ["OMIT_SHARED_CACHE"]
```

#### 4. Feature Traits

```swift
// Already enabled by default

    "ENABLE_CARRAY",        // C array extension
    "ENABLE_FTS5",          // Full-text search v5
    "ENABLE_MATH_FUNCTIONS",// Math functions

    "ENABLE_BYTECODE_VTAB",  // Bytecode virtual table
```

## Verifying Your Installation

### Basic Verification

```swift
import CSQLite

struct InstallationVerification {
    static func verify() {
        print("🔍 Verifying CSQLite installation...")
        
        // 1. Check version
        let version = String(cString: sqlite3_libversion())
        let sourceId = String(cString: sqlite3_sourceid())
        print("✅ SQLite version: \(version)")
        print("✅ Source ID: \(sourceId)")
        
        // 2. Check threading mode
        let threadSafe = sqlite3_threadsafe()
        print("✅ Thread-safe mode: \(threadSafe) (0=single, 1=serialized, 2=multi)")
        
        // 3. Test database operations
        testDatabaseOperations()
    }
    
    private static func testDatabaseOperations() {
        var db: OpaquePointer?
        
        // Open in-memory database
        guard sqlite3_open(":memory:", &db) == SQLITE_OK else {
            print("❌ Failed to open database")
            return
        }
        defer { sqlite3_close(db) }
        
        // Create a table
        let createSQL = "CREATE TABLE test(id INTEGER PRIMARY KEY, name TEXT)"
        var errMsg: UnsafeMutablePointer<CChar>?
        
        guard sqlite3_exec(db, createSQL, nil, nil, &errMsg) == SQLITE_OK else {
            let error = errMsg.map { String(cString: $0) } ?? "Unknown error"
            print("❌ Failed to create table: \(error)")
            sqlite3_free(errMsg)
            return
        }
        
        // Insert data
        let insertSQL = "INSERT INTO test (name) VALUES ('Hello CSQLite')"
        guard sqlite3_exec(db, insertSQL, nil, nil, &errMsg) == SQLITE_OK else {
            let error = errMsg.map { String(cString: $0) } ?? "Unknown error"
            print("❌ Failed to insert: \(error)")
            sqlite3_free(errMsg)
            return
        }
        
        // Query data
        var stmt: OpaquePointer?
        let querySQL = "SELECT name FROM test"
        
        guard sqlite3_prepare_v2(db, querySQL, -1, &stmt, nil) == SQLITE_OK else {
            print("❌ Failed to prepare statement")
            return
        }
        defer { sqlite3_finalize(stmt) }
        
        if sqlite3_step(stmt) == SQLITE_ROW {
            let name = String(cString: sqlite3_column_text(stmt, 0))
            print("✅ Successfully retrieved data: '\(name)'")
        } else {
            print("❌ No data retrieved")
        }
        
        print("🎉 CSQLite is working correctly!")
    }
}

// Run verification
InstallationVerification.verify()
```

### Checking Enabled Features

```swift
import CSQLite

struct FeatureChecker {
    static func checkFeatures() {
        var db: OpaquePointer?
        guard sqlite3_open(":memory:", &db) == SQLITE_OK else {
            print("❌ Failed to open database")
            return
        }
        defer { sqlite3_close(db) }
        
        let features: [(String, String)] = [
            ("FTS5", "SELECT sqlite3_compileoption_used('SQLITE_ENABLE_FTS5')"),
            ("Math Functions", "SELECT sqlite3_compileoption_used('SQLITE_ENABLE_MATH_FUNCTIONS')"),
            ("R*Tree", "SELECT sqlite3_compileoption_used('SQLITE_ENABLE_RTREE')"),
            ("Session", "SELECT sqlite3_compileoption_used('SQLITE_ENABLE_SESSION')"),
            ("GeoPoly", "SELECT sqlite3_compileoption_used('SQLITE_ENABLE_GEOPOLY')"),
            ("DQS=0", "SELECT sqlite3_compileoption_used('SQLITE_DQS=0')"),
            ("Threadsafe", "SELECT sqlite3_compileoption_used('SQLITE_THREADSAFE=0')")
        ]
        
        print("📊 Enabled Features:")
        for (name, sql) in features {
            var stmt: OpaquePointer?
            defer { sqlite3_finalize(stmt) }
            
            if sqlite3_prepare_v2(db, sql, -1, &stmt, nil) == SQLITE_OK {
                if sqlite3_step(stmt) == SQLITE_ROW {
                    let isEnabled = sqlite3_column_int(stmt, 0) != 0
                    print("  \(isEnabled ? "✅" : "❌") \(name)")
                }
            }
        }
    }
}

// Check features
FeatureChecker.checkFeatures()
```

### Initial Setup in App

```swift
// main.swift or App.swift
import CSQLite

class Application {
    private let databaseService: DatabaseService
    
    init() throws {
        // Configure database
        let config = DatabaseConfiguration(
            path: "app.db",
            journalMode: .wal,
            synchronous: .normal,
            cacheSize: 10000,
            foreignKeys: true
        )
        
        // Create service
        self.databaseService = try DatabaseService(configuration: config)
        
        // Run migrations
        try setupDatabase()
    }
    
    private func setupDatabase() throws {
        let migrationManager = MigrationManager(connection: databaseService.connection)
        
        // Register migrations
        migrationManager.register(Migration_001_CreateUsers())
        migrationManager.register(Migration_002_CreatePosts())
        
        // Run migrations
        try migrationManager.migrate()
    }
}

// Bootstrap
let app = try Application()
```

## Common Setup Issues and Solutions

### Issue 1: Missing SQLite Header

```
❌ Header 'sqlite3.h' not found
```

**Solution:**
```swift
// Make sure you're importing the module correctly
import CSQLite  // ✅ Correct
// not:
// import SQLite  // ❌ Wrong
// #include <sqlite3.h>  // ❌ Not needed with Swift
```

### Issue 2: Linking Errors

```
Undefined symbols for architecture x86_64: "_sqlite3_open"
```

**Solution:**
```swift
// Package.swift - Ensure target dependency
.target(
    name: "MyApp",
    dependencies: ["CSQLite"]  // ✅ Must be included
)
```

### Issue 3: Package Trait Not Applied

```
swift: error: unknown option '--traits'
```

**Solution:** Traits require Swift 6.1 or later:
```bash
swift --version
# Need at least: Swift version 6.1 or later
```

### Issue 4: Threading Issues

```
Database is locked
```

**Solution:** 
1. Check your threading mode configuration
2. Use `sqlite3_busy_timeout()` for handling locks

```swift
func configureBusyHandler(db: OpaquePointer?) {
    // Wait up to 5 seconds for locks
    sqlite3_busy_timeout(db, 5000)
}
```

```swift
// main.swift - Full setup example
import CSQLite
import Foundation

class DatabaseManager {
    private var db: OpaquePointer?
    private let queue = DispatchQueue(label: "database.queue", qos: .userInitiated)
    
    init() throws {
        try setupDatabase()
    }
    
    private func setupDatabase() throws {
        let path = DatabasePath.defaultPath()
        print("📁 Database path: \(path)")
        
        // Open with error handling
        let result = sqlite3_open(path, &db)
        
        switch result {
        case SQLITE_OK:
            print("✅ Database opened successfully")
            try configureDatabase()
            try createTables()
            
        case SQLITE_CANTOPEN:
            throw DatabaseError.cannotOpen("Cannot open database at \(path)")
            
        default:
            let error = String(cString: sqlite3_errmsg(db))
            throw DatabaseError.openFailed("\(error) (code: \(result))")
        }
    }
    
    private func configureDatabase() throws {
        guard let db = db else { return }
        
        // Enable foreign keys
        try execute("PRAGMA foreign_keys = ON")
        
        // Set busy timeout
        sqlite3_busy_timeout(db, 5000)
        
        // Configure WAL mode for better concurrency
        try execute("PRAGMA journal_mode = WAL")
        
        // Optimize cache
        try execute("PRAGMA cache_size = 10000")
        
        print("🔧 Database configured successfully")
    }
    
    private func createTables() throws {
        // Users table
        try execute("""
            CREATE TABLE IF NOT EXISTS users (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                username TEXT UNIQUE NOT NULL,
                email TEXT UNIQUE NOT NULL,
                created_at INTEGER DEFAULT (strftime('%s', 'now')),
                updated_at INTEGER DEFAULT (strftime('%s', 'now'))
            )
        """)
        
        // Sessions table for change tracking
        try execute("""
            CREATE TABLE IF NOT EXISTS sessions (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                user_id INTEGER NOT NULL,
                token TEXT NOT NULL,
                expires_at INTEGER NOT NULL,
                FOREIGN KEY (user_id) REFERENCES users(id) ON DELETE CASCADE
            )
        """)
        
        print("📊 Tables created/verified")
    }
    
    private func execute(_ sql: String) throws {
        guard let db = db else {
            throw DatabaseError.notConnected
        }
        
        var errMsg: UnsafeMutablePointer<CChar>?
        defer { sqlite3_free(errMsg) }
        
        let result = sqlite3_exec(db, sql, nil, nil, &errMsg)
        
        guard result == SQLITE_OK else {
            let error = errMsg.map { String(cString: $0) } ?? "Unknown error"
            throw DatabaseError.executeFailed(error)
        }
    }
    
    deinit {
        if let db = db {
            sqlite3_close(db)
            print("🗑️ Database closed")
        }
    }
}

enum DatabaseError: LocalizedError {
    case cannotOpen(String)
    case openFailed(String)
    case notConnected
    case executeFailed(String)
    
    var errorDescription: String? {
        switch self {
        case .cannotOpen(let msg): return "Cannot open database: \(msg)"
        case .openFailed(let msg): return "Failed to open database: \(msg)"
        case .notConnected: return "Database not connected"
        case .executeFailed(let msg): return "Execution failed: \(msg)"
        }
    }
}

// Run the setup
do {
    let db = try DatabaseManager()
    print("🎉 Database setup complete!")
} catch {
    print("❌ Setup failed: \(error.localizedDescription)")
}
```
